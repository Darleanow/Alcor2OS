434static inline void blend_glyph_row(
435 volatile u32 *dst, const u8 *src, u32 n, u32 bypp, u32 fg_r, u32 fg_g,
436 u32 fg_b, u32 bg_r, u32 bg_g, u32 bg_b, u32 fg_pk, u32 bg_pk
437)
438{
439 if(!dst || !src)
440 return;
441 if(bypp == 1u) {
442 for(u32 gx = 0; gx < n; gx++) {
443 u32 a = src[gx];
444 if(!a) {
445 dst[gx] = bg_pk;
446 continue;
447 }
448 if(a == ALPHA_OPAQUE) {
449 dst[gx] = fg_pk;
450 continue;
451 }
452 u32 inv = ALPHA_OPAQUE - a;
453 dst[gx] = BGRA_OPAQUE_ALPHA |
454 (((fg_r * a + bg_r * inv + ALPHA_ROUND_BIAS) >> BITS_PER_BYTE)
455 << BGRA_RED_SHIFT) |
456 (((fg_g * a + bg_g * inv + ALPHA_ROUND_BIAS) >> BITS_PER_BYTE)
457 << BGRA_GREEN_SHIFT) |
458 ((fg_b * a + bg_b * inv + ALPHA_ROUND_BIAS) >> BITS_PER_BYTE);
459 }
460 } else {
461 for(u32 gx = 0; gx < n; gx++) {
462 const u8 *px = src + (size_t)gx * bypp;
463 u32 a = (bypp == FB_BYTES_PER_PIXEL_32) ? (u32)px[ATLAS_RGBA_ALPHA_BYTE]
464 : (u32)px[ATLAS_GRAY_ALPHA_BYTE];
465 if(!a) {
466 dst[gx] = bg_pk;
467 continue;
468 }
469 if(a == ALPHA_OPAQUE) {
470 dst[gx] = fg_pk;
471 continue;
472 }
473 u32 inv = ALPHA_OPAQUE - a;
474 dst[gx] = BGRA_OPAQUE_ALPHA |
475 (((fg_r * a + bg_r * inv + ALPHA_ROUND_BIAS) >> BITS_PER_BYTE)
476 << BGRA_RED_SHIFT) |
477 (((fg_g * a + bg_g * inv + ALPHA_ROUND_BIAS) >> BITS_PER_BYTE)
478 << BGRA_GREEN_SHIFT) |
479 ((fg_b * a + bg_b * inv + ALPHA_ROUND_BIAS) >> BITS_PER_BYTE);
480 }
481 }
482}
